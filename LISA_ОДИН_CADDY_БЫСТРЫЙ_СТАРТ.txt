✨ ВАРИАНТ 1: АВТОМАТИЧЕСКАЯ УСТАНОВКА (ОДНА КОМАНДА)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Скопируйте и запустите эту ОДНУ команду:

curl -sSL "https://raw.githubusercontent.com/ankor1403/ankor1403/claude/audit-dependencies-mkbp0op9n8ha6qpk-Gt2Dm/install_lisa_unified_caddy.sh" | bash

Скрипт автоматически:
  ✓ Создаст бэкапы
  ✓ Остановит Telegram сервисы
  ✓ Удалит старый Caddy
  ✓ Установит Lisa
  ✓ Настроит единый Caddy для всех
  ✓ Проверит работу

Время: ~15 минут (большую часть времени Lisa устанавливается)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🔧 ВАРИАНТ 2: РУЧНАЯ УСТАНОВКА (ПОШАГОВО)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Если хотите контролировать каждый шаг, выполняйте команды блоками:


═══════════════════════════════════════════════════════════════════════════
ШАГ 1: СОЗДАТЬ БЭКАПЫ (5 минут)
═══════════════════════════════════════════════════════════════════════════

mkdir -p ~/backups/$(date +%Y%m%d_%H%M%S) && cd ~/backups/$(date +%Y%m%d_%H%M%S)
docker exec wappi-caddy cat /etc/caddy/Caddyfile > caddy_old_config.txt
cp -r /opt/telegram_multi_service /opt/telegram_multi_service.backup
cp -r /opt/messengers /opt/messengers.backup
cp -r /opt/wappi /opt/wappi.backup
docker ps > docker_containers_before.txt
echo "✓ Бэкапы созданы!"


═══════════════════════════════════════════════════════════════════════════
ШАГ 2: ОСТАНОВИТЬ СЕРВИСЫ (2 минуты)
═══════════════════════════════════════════════════════════════════════════

cd /opt/telegram_multi_service && docker-compose stop
cd /opt/messengers && docker-compose stop telegram-service
cd /opt/wappi && docker-compose down
echo "✓ Старые сервисы остановлены!"


═══════════════════════════════════════════════════════════════════════════
ШАГ 3: УСТАНОВИТЬ LISA (10 минут)
═══════════════════════════════════════════════════════════════════════════

apt update && apt install -y git python3 python3-pip
cd ~ && git clone https://github.com/shorin-nikita/lisa.git && cd lisa
python3 CTAPT.py

⚠️ ОТВЕТЬТЕ НА ВОПРОСЫ:
  • Домен для N8N: n8n.afonin-lisa.ru
  • Домен для Supabase: db.afonin-lisa.ru
  • Домен для Ollama: ai.afonin-lisa.ru
  • Email для SSL: admin@afonin-lisa.ru
  • Прокси: (просто Enter)

⏳ Ждите 5-10 минут...


═══════════════════════════════════════════════════════════════════════════
ШАГ 4: ЗАПУСТИТЬ TELEGRAM СЕРВИСЫ (1 минута)
═══════════════════════════════════════════════════════════════════════════

cd /opt/telegram_multi_service && docker-compose start
cd /opt/messengers && docker-compose start telegram-service
echo "✓ Telegram сервисы запущены!"


═══════════════════════════════════════════════════════════════════════════
ШАГ 5: ПОДКЛЮЧИТЬ К СЕТИ LISA (1 минута)
═══════════════════════════════════════════════════════════════════════════

# Найти имя сети (запомните результат!):
docker network ls | grep lisa

# Подключить (замените lisa_default на имя из предыдущей команды):
docker network connect lisa_default telegram-multi-service
docker network connect lisa_default telegram-service
echo "✓ Подключено к сети Lisa!"


═══════════════════════════════════════════════════════════════════════════
ШАГ 6: НАСТРОИТЬ CADDYFILE (5 минут)
═══════════════════════════════════════════════════════════════════════════

# Скачать готовый Caddyfile:
cd ~/lisa
curl -sSL "https://raw.githubusercontent.com/ankor1403/ankor1403/claude/audit-dependencies-mkbp0op9n8ha6qpk-Gt2Dm/Caddyfile.unified" -o Caddyfile.new

# Найти имя контейнера Caddy (запомните!):
docker ps | grep caddy

# Скопировать в контейнер (замените lisa_caddy_1 на имя из предыдущей команды):
docker cp ~/lisa/Caddyfile.new lisa_caddy_1:/etc/caddy/Caddyfile

# Проверить синтаксис:
docker exec lisa_caddy_1 caddy validate --config /etc/caddy/Caddyfile

# Перезагрузить Caddy:
docker exec lisa_caddy_1 caddy reload --config /etc/caddy/Caddyfile

echo "✓ Caddy настроен!"

⚠️ НЕ ЗАБУДЬТЕ НАСТРОИТЬ DNS!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

В панели управления вашим доменом создайте A-записи:

  n8n.afonin-lisa.ru      → A → 83.222.25.94
  db.afonin-lisa.ru       → A → 83.222.25.94
  ai.afonin-lisa.ru       → A → 83.222.25.94

Без этого SSL не будет работать для новых доменов!

🆘 ЕСЛИ ЧТО-ТО НЕ РАБОТАЕТ

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Проблема: Домены не открываются
Решение:
  # Проверить DNS:
  nslookup n8n.afonin-lisa.ru
  # Проверить логи Caddy:
  docker logs --tail=50 lisa_caddy_1

Проблема: Telegram не работает
Решение:
  # Проверить подключение к сети:
  docker network inspect lisa_default | grep telegram
  # Если нет - подключить:
  docker network connect lisa_default telegram-multi-service

Проблема: SSL не работает
Решение:
  # Убедитесь что DNS настроены правильно
  # Перезагрузите Caddy:
  docker exec lisa_caddy_1 caddy reload --config /etc/caddy/Caddyfile

Проблема: Не знаю пароли Lisa
Решение:
  cd ~/lisa && cat .env | grep PASSWORD
